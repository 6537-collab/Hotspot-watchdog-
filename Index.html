package com.hotspot.watchdog.service

import android.app.*
import android.content.*
import android.net.ConnectivityManager
import android.net.NetworkCapabilities
import android.os.*
import androidx.core.app.NotificationCompat
import com.hotspot.watchdog.R
import com.hotspot.watchdog.net.FcmPushManager
import kotlinx.coroutines.*
import java.io.File
import java.io.BufferedReader
import java.io.FileReader

class WatchdogService : Service() {

    private val serviceScope = CoroutineScope(Dispatchers.IO + SupervisorJob())
    private var lastInternetState: Boolean? = null
    private var lastChargingState: String? = null
    private val knownDevices = mutableSetOf<String>()

    override fun onCreate() {
        super.onCreate()
        startForeground(NOTIFICATION_ID, createNotification())
        registerBatteryReceiver()
        startMonitoringLoop()
    }

    private fun startMonitoringLoop() {
        serviceScope.launch {
            while (isActive) {
                checkInternetConnectivity()
                checkHotspotDevices()
                delay(30_000) // Low power 30-second interval check
            }
        }
    }

    // Rule 1: Instant Internet Connection Check
    private fun checkInternetConnectivity() {
        val cm = getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
        val activeNetwork = cm.activeNetwork
        val caps = cm.getNetworkCapabilities(activeNetwork)
        val isOnline = caps?.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET) == true

        if (lastInternetState != null && lastInternetState != isOnline) {
            if (!isOnline) {
                FcmPushManager.sendPushAlert("🚨 Internet Offline", "Hotspot phone lost connectivity!")
            } else {
                FcmPushManager.sendPushAlert("🌐 Internet Restored", "Hotspot internet connection is back online ✅")
            }
        }
        lastInternetState = isOnline
    }

    // Rule 2: Battery & Charging Broadcast Listener
    private void registerBatteryReceiver() {
        val filter = IntentFilter(Intent.ACTION_BATTERY_CHANGED)
        registerReceiver(object : BroadcastReceiver() {
            override fun onReceive(context: Context?, intent: Intent?) {
                intent?.let {
                    val status = it.getIntExtra(BatteryManager.EXTRA_STATUS, -1)
                    val isCharging = status == BatteryManager.BATTERY_STATUS_CHARGING ||
                                     status == BatteryManager.BATTERY_STATUS_FULL
                    val level = it.getIntExtra(BatteryManager.EXTRA_LEVEL, -1)
                    val scale = it.getIntExtra(BatteryManager.EXTRA_SCALE, -1)
                    val batteryPct = (level * 100 / scale.toFloat()).toInt()

                    val currentState = when {
                        status == BatteryManager.BATTERY_STATUS_FULL -> "FULL"
                        isCharging -> "CHARGING"
                        else -> "UNPLUGGED"
                    }

                    if (lastChargingState != null && lastChargingState != currentState) {
                        if (currentState == "UNPLUGGED" && batteryPct < 100) {
                            FcmPushManager.sendPushAlert("⚠️ Hotspot Unplugged", "Battery at $batteryPct%. Please plug in!")
                        } else if (currentState == "CHARGING") {
                            FcmPushManager.sendPushAlert("🔌 Charging Started", "Hotspot phone is plugged in.")
                        } else if (currentState == "FULL") {
                            FcmPushManager.sendPushAlert("🔋 Battery Full", "Battery reached 100%.")
                        }
                    }
                    lastChargingState = currentState
                }
            }
        }, filter)
    }

    // Rule 3: Hotspot Devices ARP Table Reader (/proc/net/arp)
    private fun checkHotspotDevices() {
        val currentDevices = mutableSetOf<String>()
        try {
            val br = BufferedReader(FileReader("/proc/net/arp"))
            var line: String?
            while (br.readLine().also { line = it } != null) {
                val tokens = line!!.split("\\s+".toRegex())
                if (tokens.size >= 4 && tokens[0] != "IP") {
                    val ip = tokens[0]
                    val mac = tokens[3]
                    if (mac != "00:00:00:00:00:00") {
                        currentDevices.add("$ip ($mac)")
                    }
                }
            }
            br.close()

            // Check joined devices
            val joined = currentDevices - knownDevices
            joined.forEach { device ->
                FcmPushManager.sendPushAlert("📱 Device Joined", "New client connected: $device")
            }

            // Check disconnected devices
            val left = knownDevices - currentDevices
            left.forEach { device ->
                FcmPushManager.sendPushAlert("📱 Device Left", "Client disconnected: $device")
            }

            knownDevices.clear()
            knownDevices.addAll(currentDevices)
        } catch (e: Exception) {
            e.printStackTrace()
        }
    }

    private fun createNotification(): Notification {
        val channelId = "watchdog_service"
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(channelId, "Hotspot Watchdog Foreground", NotificationManager.IMPORTANCE_LOW)
            getSystemService(NotificationManager::class.java).createNotificationChannel(channel)
        }
        return NotificationCompat.Builder(this, channelId)
            .setContentTitle("Hotspot Watchdog Active")
            .setContentText("Monitoring network, battery, and connected clients...")
            .setSmallIcon(R.drawable.ic_shield)
            .setOngoing(true)
            .build()
    }

    override fun onBind(intent: Intent?): IBinder? = null

    companion object {
        const val NOTIFICATION_ID = 1001
    }
}
package com.hotspot.watchdog.net

import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.net.HttpURLConnection
import java.net.URL
import org.json.JSONObject

object FcmPushManager {
    
    private var targetFcmToken: String? = null

    fun setTargetToken(token: String) {
        this.targetFcmToken = token
    }

    suspend fun sendPushAlert(title: String, body: String) = withContext(Dispatchers.IO) {
        val token = targetFcmToken ?: return@withContext
        
        try {
            val url = URL("https://fcm.googleapis.com/fcm/send")
            val conn = url.openConnection() as HttpURLConnection
            conn.requestMethod = "POST"
            conn.setRequestProperty("Authorization", "key=YOUR_FCM_SERVER_KEY")
            conn.setRequestProperty("Content-Type", "application/json")
            conn.doOutput = true

            val payload = JSONObject().apply {
                put("to", token)
                put("priority", "high")
                put("notification", JSONObject().apply {
                    put("title", title)
                    put("body", body)
                    put("sound", "default")
                })
                put("data", JSONObject().apply {
                    put("timestamp", System.currentTimeMillis())
                })
            }

            val os = conn.outputStream
            os.write(payload.toString().toByteArray())
            os.flush()
            os.close()

            val responseCode = conn.responseCode
            println("FCM Push Sent. Response Code: $responseCode")
        } catch (e: Exception) {
            e.printStackTrace()
        }
    }
}
package com.hotspot.watchdog.pair

import android.graphics.Bitmap
import com.google.zxing.BarcodeFormat
import com.google.zxing.qrcode.QRCodeWriter
import org.json.JSONObject

object PairManager {

    data class PairPayload(
        val deviceName: String,
        val fcmToken: String,
        val createdTimestamp: Long
    )

    fun generatePairQrJson(deviceName: String, fcmToken: String): String {
        return JSONObject().apply {
            put("name", deviceName)
            put("token", fcmToken)
            put("ts", System.currentTimeMillis())
        }.toString()
    }

    fun parsePairQrJson(qrRawText: String): PairPayload? {
        return try {
            val json = JSONObject(qrRawText)
            PairPayload(
                deviceName = json.getString("name"),
                fcmToken = json.getString("token"),
                createdTimestamp = json.getLong("ts")
            )
        } catch (e: Exception) {
            null
        }
    }
}
package com.hotspot.watchdog.receiver

import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import android.os.Build
import com.hotspot.watchdog.service.WatchdogService

class BootReceiver : BroadcastReceiver() {
    override fun onReceive(context: Context?, intent: Intent?) {
        if (intent?.action == Intent.ACTION_BOOT_COMPLETED ||
            intent?.action == Intent.ACTION_MY_PACKAGE_REPLACED) {
            
            context?.let { ctx ->
                val serviceIntent = Intent(ctx, WatchdogService::class.java)
                if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                    ctx.startForegroundService(serviceIntent)
                } else {
                    ctx.startService(serviceIntent)
                }
            }
        }
    }
}
